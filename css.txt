/* =========================================
   START QUIZ
========================================= */

function startQuiz() {

  currentQuestion = 0;
  quizAnswers = {};

  renderQuiz();

  window.scrollTo({
    top: 0,
    behavior: "smooth"
  });
}


/* =========================================
   RENDER QUIZ
========================================= */

function renderQuiz() {

  const main = document.querySelector("main");

  if (!main) return;

  const question = questions[currentQuestion];

  main.innerHTML = `

    <section class="quiz" style="display:flex;">

      <div class="quiz-inner">

        <p class="eyebrow">
          ${question.eyebrow}
        </p>

        <div class="quiz-progress">

          <span>
            ${String(currentQuestion + 1).padStart(2, "0")}
          </span>

          <div class="progress-line"></div>

          <span>
            ${String(questions.length).padStart(2, "0")}
          </span>

        </div>

        <h2>
          ${question.title}
        </h2>

        <p class="quiz-description">
          ${question.description}
        </p>

        <div class="quiz-options">

          ${question.options.map((option, index) => `

            <button
              class="quiz-option"
              onclick="selectAnswer('${option[0]}')"
            >

              <span>
                ${String(index + 1).padStart(2, "0")}
              </span>

              ${option[1]}

            </button>

          `).join("")}

        </div>

      </div>

    </section>

  `;

  window.scrollTo({
    top: 0,
    behavior: "smooth"
  });
}


/* =========================================
   SELECT ANSWER
========================================= */

function selectAnswer(answer) {

  const question = questions[currentQuestion];

  quizAnswers[question.key] = answer;

  currentQuestion++;

  if (currentQuestion < questions.length) {

    renderQuiz();

  } else {

    showResults();

  }
}


/* =========================================
   CALCULATE RECOMMENDATIONS
========================================= */

function calculateRecommendations() {

  return gifts.map(gift => {

    let score = 0;

    const answers = quizAnswers;


    /* Relationship */

    if (
      gift.relationships &&
      gift.relationships.includes(answers.relationship)
    ) {
      score += 12;
    }


    /* Personality */

    if (
      gift.personalities &&
      gift.personalities.includes(answers.personality)
    ) {
      score += 10;
    }


    /* Interest */

    if (
      gift.interests &&
      gift.interests.includes(answers.interest)
    ) {
      score += 18;
    }


    /* Intimacy */

    if (
      gift.intimacy &&
      gift.intimacy.includes(answers.intimacy)
    ) {
      score += 9;
    }


    /* Budget */

    if (gift.budget === answers.budget) {

      score += 20;

    } else {

      if (
        answers.budget === "under100" &&
        gift.price <= 150
      ) {
        score += 7;
      }

      if (
        answers.budget === "100to300" &&
        gift.price <= 500
      ) {
        score += 7;
      }

      if (
        answers.budget === "300to1000" &&
        gift.price <= 1200
      ) {
        score += 5;
      }

    }


    /* Special relationship */

    if (
      answers.relationship === "partner" &&
      answers.intimacy === "very-personal"
    ) {
      if (
        gift.relationships.includes("partner") &&
        gift.intimacy.includes("very-personal")
      ) {
        score += 8;
      }
    }


    if (
      answers.relationship === "special" &&
      answers.intimacy === "very-personal"
    ) {
      if (gift.intimacy.includes("very-personal")) {
        score += 10;
      }
    }


    /* Colleague */

    if (
      answers.relationship === "colleague" &&
      answers.intimacy === "safe"
    ) {

      if (
        gift.relationships.includes("colleague")
      ) {
        score += 7;
      }

    }


    /* Creative */

    if (
      answers.personality === "creative" &&
      ["art", "home"].includes(gift.category)
    ) {
      score += 7;
    }


    /* Intellectual */

    if (
      answers.personality === "intellectual" &&
      ["books", "art"].includes(gift.category)
    ) {
      score += 7;
    }


    /* Adventurous */

    if (
      answers.personality === "adventurous" &&
      gift.category === "experience"
    ) {
      score += 10;
    }


    /* Minimalist */

    if (
      answers.personality === "minimalist" &&
      ["home", "fashion", "art"].includes(gift.category)
    ) {
      score += 5;
    }


    /* Urgency */

    if (
      answers.urgency === "today" &&
      gift.category === "experience"
    ) {
      score -= 8;
    }


    /* Low budget protection */

    if (
      answers.budget === "under100" &&
      gift.price > 150
    ) {
      score -= 18;
    }


    if (
      answers.budget === "100to300" &&
      gift.price > 500
    ) {
      score -= 12;
    }


    return {
      ...gift,
      score
    };

  }).sort((a, b) => b.score - a.score);
}


/* =========================================
   DIVERSIFY CATEGORIES
========================================= */

function diversifyRecommendations(items, limit) {

  const result = [];
  const categoryCounts = {};

  for (const item of items) {

    const category = item.category;

    if (!categoryCounts[category]) {
      categoryCounts[category] = 0;
    }

    if (categoryCounts[category] >= 4) {
      continue;
    }

    result.push(item);

    categoryCounts[category]++;

    if (result.length >= limit) {
      break;
    }
  }

  return result;
}

/* =========================================
   FILTER BUTTON
========================================= */

function createFilterButton(
  category,
  label,
  active = false
) {

  return `

    <button
      class="results-filter ${active ? "active" : ""}"
      onclick="filterRecommendations('${category}', this)"
    >
      ${label}
    </button>

  `;
}


/* =========================================
   FILTER RECOMMENDATIONS
========================================= */

function filterRecommendations(
  category,
  button
) {

  currentCategory = category;

  document
    .querySelectorAll(".results-filter")
    .forEach(item => {
      item.classList.remove("active");
    });

  if (button) {
    button.classList.add("active");
  }

  visibleRecommendations = 12;

  renderRecommendations();
}


/* =========================================
   RENDER RECOMMENDATIONS
========================================= */

function renderRecommendations() {

  const grid = document.getElementById(
    "recommendations-grid"
  );

  const loadMore = document.getElementById(
    "load-more-wrapper"
  );

  const count = document.getElementById(
    "selection-count"
  );

  if (!grid) return;


  let items = currentRecommendations;


  if (currentCategory !== "all") {

    items = currentRecommendations.filter(
      gift => gift.category === currentCategory
    );

  }


  const visibleItems = items.slice(
    0,
    visibleRecommendations
  );


  grid.innerHTML = visibleItems.map(
    (gift, index) => {

      const saved = savedGifts.includes(gift.id);

      return `

        <article class="recommendation-card">

          <div class="recommendation-image">

            <img
              src="${gift.image}"
              alt="${escapeHTML(gift.name)}"
              loading="lazy"
            >

            <button
              class="save-button"
              onclick="saveGift(${gift.id})"
              aria-label="Save gift"
            >
              ${saved ? "¦" : "?"}
            </button>

          </div>


          <div class="recommendation-content">

            <div class="recommendation-number">
              ${String(index + 1).padStart(2, "0")}
            </div>

            <div class="recommendation-brand">
              ${escapeHTML(gift.brand)}
            </div>

            <div class="recommendation-info">

              <h3>
                ${escapeHTML(gift.name)}
              </h3>

              <div class="recommendation-price">
                ˆ${gift.price.toLocaleString()}
              </div>

              <p class="recommendation-reason">
                ${escapeHTML(gift.description)}
              </p>

              <a
                href="#"
                class="view-object"
                onclick="viewGift(${gift.id}); return false;"
              >
                View object >
              </a>

            </div>

          </div>

        </article>

      `;

    }
  ).join("");


  if (count) {

    count.textContent =
      `${visibleItems.length} OF ${items.length} OBJECTS`;

  }


  if (loadMore) {

    if (visibleItems.length < items.length) {

      loadMore.innerHTML = `

        <button
          class="load-more-button"
          onclick="loadMoreRecommendations()"
        >
          View more selections
        </button>

      `;

    } else {

      loadMore.innerHTML = "";

    }

  }

}


/* =========================================
   LOAD MORE
========================================= */

function loadMoreRecommendations() {

  visibleRecommendations += 12;

  renderRecommendations();

}


/* =========================================
   SAVE GIFT
========================================= */

function saveGift(id) {

  if (savedGifts.includes(id)) {

    savedGifts = savedGifts.filter(
      giftId => giftId !== id
    );

  } else {

    savedGifts.push(id);

  }

  localStorage.setItem(
    "ro_saved_gifts",
    JSON.stringify(savedGifts)
  );

  renderRecommendations();

}


/* =========================================
   VIEW OBJECT
========================================= */

function viewGift(id) {

  const gift = gifts.find(
    item => item.id === id
  );

  if (!gift) return;

  alert(
    `${gift.name}\n\n${gift.brand}\nˆ${gift.price}\n\n${gift.description}`
  );

}

/* =========================================
   ESCAPE HTML
========================================= */

function escapeHTML(text) {

  return String(text)
    .replace(/&/g, "&amp;")
    .replace(/</g, "&lt;")
    .replace(/>/g, "&gt;")
    .replace(/"/g, "&quot;")
    .replace(/'/g, "&#039;");

}
